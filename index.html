only allow processing throws in 301 or 501 modes if self.game_var.get() not in (301, 501): return try: score = int(self.score_entry.get()) if score == 0: self.root.destroy() return player = self.current_player prev = self.player_scores[player] # validation if score < 0 or score > 180 or score > prev: messagebox.showerror("Invalid Score", "Must be between 1-180 and <= remaining score") return new = prev - score # win if new == 0: # push move to history so undo could be possible before reset? we'll reset which clears history self.reset_game() return # if remaining is less than 2, treat as bust (cannot finish) if new < 2: messagebox.showinfo("Player Busted", "Player Busted") return # bust if new < 0: messagebox.showinfo("Bust!", "Score went below zero. Throw canceled.") return # valid throw: record history and apply switched = (self.players_var.get() == 2) self.history.append({'player': player, 'prev': prev, 'score': score, 'switched': switched}) self.player_scores[player] = new self.score_label.config(text=f"Player {player+1} Score: {self.player_scores[player]}") self.score_entry.delete(0, 'end') # After a valid throw, update display and switch player if two players self.update_player_display() if switched: self.switch_player() self.update_checkout_suggestions() except ValueError: messagebox.showerror("Invalid Input", "Please enter a valid number") def undo_last(self): if not self.history: messagebox.showinfo("Undo", "No moves to undo") return last = self.history.pop() typ = last.get('type', 'throw') if typ == 'throw': idx = last['player'] self.player_scores[idx] = last['prev'] self.current_player = idx elif typ == 'cricket': idx = last['player'] label = last['label'] # restore accum self.cricket_accum[idx][label] = last.get('prev_accum', 0) # subtract any score added score_added = last.get('score_added', 0) if score_added: try: self.player_scores[idx] -= score_added except Exception: pass # restore current player to the clicking player self.current_player = idx # refresh displays self.update_player_display() self.update_checkout_suggestions() self.update_player_columns() self.update_player_columns() def quit_game(self): # gracefully terminate try: self.root.destroy() except Exception: pass def on_entry_click(self, event): try: self.score_entry.focus_set() self.score_entry.select_range(0, tk.END) except Exception: pass def on_enter_key(self, event=None): # otherwise focus the entry so the user can type try: self.score_entry.focus_set() except Exception: pass def cricket_click(self, player_idx, label_idx): # Only active in cricket mode if self.game_var.get() != 0: return # increment accumulated score by label value new_acc = prev_acc + val self.cricket_accum[player_idx][label_idx] = new_acc new_hits = new_acc // val if val else 0 # determine if this click scores points: if this click produced an extra hit beyond 3 opponent_idx = 1 - player_idx opp_hits = self.cricket_accum[opponent_idx][label_idx] // val score_added = 0 if new_hits > 3 and opp_hits < 3: # this click can add one label value to player's total score_added = val try: self.player_scores[player_idx] += score_added except Exception: pass # record to history for undo (type 'cricket') self.history.append({'type': 'cricket', 'player': player_idx, 'label': label_idx, 'prev_accum': prev_acc, 'score_added': score_added}) # update displays self.update_player_display() self.update_player_columns() self.update_checkout_suggestions() # Do NOT automatically switch player on cricket clicks; Next button controls switching. # Check for cricket win condition: player has >=3 hits on all labels and has higher score try: all_closed = True for idx, val in enumerate(self.cricket_vals): hits = self.cricket_accum[player_idx][idx] // (val if val else 1) if hits < 3: all_closed = False break if all_closed: opponent = 1 - player_idx if self.player_scores[player_idx] > self.player_scores[opponent]: # mark game over so no further scoring is allowed self.reset_game() return except Exception: pass def next_player(self): # Toggle active player (used in cricket mode) self.current_player = 1 - self.current_player self.update_player_display() self.update_player_columns() def on_players_change(self): #
